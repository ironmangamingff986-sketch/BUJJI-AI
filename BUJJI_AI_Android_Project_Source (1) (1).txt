package com.bujji.ai.viewmodel

import android.app.Application
import androidx.lifecycle.AndroidViewModel
import androidx.lifecycle.viewModelScope
import com.bujji.ai.data.GeminiRepository
import com.bujji.ai.engine.OfflineCommandEngine
import com.bujji.ai.speech.OfflineSpeechManager
import com.bujji.ai.model.BujjiUiState
import com.bujji.ai.model.Message
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.asStateFlow
import kotlinx.coroutines.flow.update
import kotlinx.coroutines.launch

class BujjiViewModel(application: Application) : AndroidViewModel(application) {

    private val _uiState = MutableStateFlow(BujjiUiState())
    val uiState: StateFlow<BujjiUiState> = _uiState.asStateFlow()

    private val offlineCommandEngine = OfflineCommandEngine(application)
    private val geminiRepository = GeminiRepository()
    private var speechManager: OfflineSpeechManager? = null

    init {
        speechManager = OfflineSpeechManager(
            context = application,
            onResult = { transcript -> handleIncomingTranscript(transcript) },
            onRmsChanged = { rms ->
                val normalized = (rms / 10f).coerceIn(0f, 1f)
                _uiState.update { it.copy(audioVolume = normalized) }
            },
            onError = { error ->
                _uiState.update { it.copy(isListening = false, errorMessage = error) }
            }
        )
    }

    fun onMicrophonePermissionResult(isGranted: Boolean) {
        _uiState.update { it.copy(hasMicrophonePermission = isGranted) }
    }

    fun onInputQueryChanged(newQuery: String) {
        _uiState.update { it.copy(inputQuery = newQuery) }
    }

    fun setLanguage(lang: String) {
        _uiState.update { it.copy(language = lang) }
    }

    fun toggleListening() {
        val currentState = _uiState.value
        if (currentState.isListening) {
            speechManager?.stopListening()
            _uiState.update { it.copy(isListening = false, audioVolume = 0f) }
        } else {
            val langCode = if (currentState.language == "te") "te-IN" else "en-IN"
            speechManager?.startListening(languageCode = langCode, preferOffline = true)
            _uiState.update { it.copy(isListening = true, errorMessage = null) }
        }
    }

    private fun handleIncomingTranscript(transcript: String) {
        _uiState.update { it.copy(isListening = false, lastTranscript = transcript) }
        processQuery(transcript)
    }

    fun sendQuery(text: String = _uiState.value.inputQuery) {
        val query = text.trim()
        if (query.isEmpty()) return
        _uiState.update { it.copy(inputQuery = "") }
        processQuery(query)
    }

    private fun processQuery(query: String) {
        val currentLang = _uiState.value.language
        val userMsg = Message(sender = "user", text = query, language = currentLang)
        _uiState.update { it.copy(messages = listOf(userMsg) + it.messages) }

        // 1. Check Offline Commands
        val offlineResult = offlineCommandEngine.evaluate(query)
        if (offlineResult != null && offlineResult.isHandled) {
            val bujjiMsg = Message(
                sender = "bujji",
                text = offlineResult.responseText,
                isOnline = false,
                language = if (offlineResult.isTelugu) "te" else "en"
            )
            _uiState.update {
                it.copy(
                    messages = listOf(bujjiMsg) + it.messages,
                    lastResponse = offlineResult.responseText
                )
            }
            speechManager?.speak(offlineResult.responseText, offlineResult.isTelugu)
            return
        }

        // 2. Query Online Gemini AI
        _uiState.update { it.copy(isProcessing = true) }
        viewModelScope.launch(Dispatchers.IO) {
            val result = geminiRepository.queryGemini(query, currentLang)
            val isTe = currentLang == "te" || query.any { it in '\u0C00'..'\u0C7F' }

            result.onSuccess { replyText ->
                val bujjiMsg = Message(sender = "bujji", text = replyText, isOnline = true, language = if (isTe) "te" else "en")
                _uiState.update {
                    it.copy(isProcessing = false, messages = listOf(bujjiMsg) + it.messages, lastResponse = replyText)
                }
                speechManager?.speak(replyText, isTe)
            }.onFailure { error ->
                val fallbackText = if (isTe) "కనెక్టివిటీ సమస్య వచ్చింది బాస్." else "Neural link momentarily disrupted."
                val bujjiMsg = Message(sender = "bujji", text = fallbackText, isOnline = false, language = if (isTe) "te" else "en")
                _uiState.update {
                    it.copy(isProcessing = false, messages = listOf(bujjiMsg) + it.messages, lastResponse = fallbackText)
                }
                speechManager?.speak(fallbackText, isTe)
            }
        }
    }
}

package com.bujji.ai.engine

import android.content.Context
import android.hardware.camera2.CameraManager
import android.os.BatteryManager
import java.text.SimpleDateFormat
import java.util.Date
import java.util.Locale

class OfflineCommandEngine(private val context: Context) {

    private val cameraManager = context.getSystemService(Context.CAMERA_SERVICE) as? CameraManager

    data class CommandResult(
        val isHandled: Boolean,
        val responseText: String,
        val isTelugu: Boolean
    )

    fun evaluate(query: String): CommandResult? {
        val input = query.lowercase(Locale.ROOT)
        val isTelugu = input.any { it in '\u0C00'..'\u0C7F' }

        // Battery
        if (input.contains("battery") || input.contains("చార్జింగ్") || input.contains("బ్యాటరీ")) {
            val batteryManager = context.getSystemService(Context.BATTERY_SERVICE) as? BatteryManager
            val level = batteryManager?.getIntProperty(BatteryManager.BATTERY_PROPERTY_CAPACITY) ?: 88
            val reply = if (isTelugu) "ఆర్క్ రియాక్టర్ పవర్ $level% వద్ద ఉంది బాస్!" else "Arc Reactor core is at $level%."
            return CommandResult(true, reply, isTelugu)
        }

        // Torch / Flashlight
        if (input.contains("torch on") || input.contains("flashlight on") || input.contains("లైట్ వెలిగించు")) {
            toggleTorch(true)
            val reply = if (isTelugu) "ఫ్లాష్‌లైట్ ఆన్ చేశాను బాస్!" else "High-beam torch engaged."
            return CommandResult(true, reply, isTelugu)
        }
        if (input.contains("torch off") || input.contains("flashlight off") || input.contains("లైట్ ఆపు")) {
            toggleTorch(false)
            val reply = if (isTelugu) "ఫ్లాష్‌లైట్ ఆఫ్ చేశాను బాస్." else "Torch deactivated."
            return CommandResult(true, reply, isTelugu)
        }

        // Time
        if (input.contains("time") || input.contains("సమయం")) {
            val time = SimpleDateFormat("hh:mm a", Locale.getDefault()).format(Date())
            val reply = if (isTelugu) "ప్రస్తుత సమయం $time బాస్!" else "The current local time is $time."
            return CommandResult(true, reply, isTelugu)
        }

        return null
    }

    private fun toggleTorch(enable: Boolean) {
        try {
            cameraManager?.let { cm ->
                val cameraId = cm.cameraIdList.firstOrNull() ?: return
                cm.setTorchMode(cameraId, enable)
            }
        } catch (_: Exception) {}
    }
}

package com.bujji.ai.speech

import android.content.Context
import android.content.Intent
import android.os.Bundle
import android.speech.RecognitionListener
import android.speech.RecognizerIntent
import android.speech.SpeechRecognizer
import android.speech.tts.TextToSpeech
import java.util.Locale

class OfflineSpeechManager(
    private val context: Context,
    private val onResult: (String) -> Unit,
    private val onRmsChanged: (Float) -> Unit,
    private val onError: (String) -> Unit
) {
    private var speechRecognizer: SpeechRecognizer? = null
    private var textToSpeech: TextToSpeech? = null
    private var ttsReady = false

    init {
        if (SpeechRecognizer.isRecognitionAvailable(context)) {
            speechRecognizer = SpeechRecognizer.createSpeechRecognizer(context).apply {
                setRecognitionListener(object : RecognitionListener {
                    override fun onReadyForSpeech(params: Bundle?) {}
                    override fun onBeginningOfSpeech() {}
                    override fun onRmsChanged(rmsdB: Float) { onRmsChanged(rmsdB) }
                    override fun onBufferReceived(buffer: ByteArray?) {}
                    override fun onEndOfSpeech() {}
                    override fun onError(error: Int) { onError("Speech code: $error") }
                    override fun onResults(results: Bundle?) {
                        val matches = results?.getStringArrayList(SpeechRecognizer.RESULTS_RECOGNITION)
                        if (!matches.isNullOrEmpty()) onResult(matches[0])
                    }
                    override fun onPartialResults(partialResults: Bundle?) {}
                    override fun onEvent(eventType: Int, params: Bundle?) {}
                })
            }
        }

        textToSpeech = TextToSpeech(context) { status ->
            if (status == TextToSpeech.SUCCESS) {
                ttsReady = true
                textToSpeech?.language = Locale("te", "IN")
            }
        }
    }

    fun startListening(languageCode: String = "te-IN", preferOffline: Boolean = true) {
        val intent = Intent(RecognizerIntent.ACTION_RECOGNIZE_SPEECH).apply {
            putExtra(RecognizerIntent.EXTRA_LANGUAGE_MODEL, RecognizerIntent.LANGUAGE_MODEL_FREE_FORM)
            putExtra(RecognizerIntent.EXTRA_LANGUAGE, languageCode)
            if (preferOffline) putExtra(RecognizerIntent.EXTRA_PREFER_OFFLINE, true)
        }
        speechRecognizer?.startListening(intent)
    }

    fun stopListening() { speechRecognizer?.stopListening() }

    fun speak(text: String, isTelugu: Boolean = false) {
        if (!ttsReady) return
        textToSpeech?.language = if (isTelugu) Locale("te", "IN") else Locale.US
        textToSpeech?.speak(text, TextToSpeech.QUEUE_FLUSH, null, "BUJJI")
    }

    fun release() {
        speechRecognizer?.destroy()
        textToSpeech?.shutdown()
    }
}

package com.bujji.ai.data

import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.withContext
import org.json.JSONObject
import java.io.OutputStreamWriter
import java.net.HttpURLConnection
import java.net.URL

class GeminiRepository(private val backendUrl: String = "https://your-bujji-api-endpoint.com") {

    suspend fun queryGemini(message: String, language: String = "auto"): Result<String> = withContext(Dispatchers.IO) {
        try {
            val url = URL("$backendUrl/api/chat")
            val conn = (url.openConnection() as HttpURLConnection).apply {
                requestMethod = "POST"
                setRequestProperty("Content-Type", "application/json")
                connectTimeout = 5000
                readTimeout = 8000
                doOutput = true
            }

            val payload = JSONObject().apply {
                put("message", message)
                put("language", language)
            }

            OutputStreamWriter(conn.outputStream).use { it.write(payload.toString()); it.flush() }

            if (conn.responseCode == 200) {
                val body = conn.inputStream.bufferedReader().use { it.readText() }
                val reply = JSONObject(body).optString("reply", "Understood, Captain.")
                Result.success(reply)
            } else {
                Result.failure(Exception("HTTP ${conn.responseCode}"))
            }
        } catch (e: Exception) {
            Result.failure(e)
        }
    }
}

package com.bujji.ai

import android.Manifest
import android.content.pm.PackageManager
import android.os.Bundle
import androidx.activity.ComponentActivity
import androidx.activity.compose.setContent
import androidx.activity.result.contract.ActivityResultContracts
import androidx.activity.viewModels
import androidx.core.content.ContextCompat
import com.bujji.ai.ui.BujjiScreen
import com.bujji.ai.viewmodel.BujjiViewModel

class MainActivity : ComponentActivity() {

    private val viewModel: BujjiViewModel by viewModels()

    private val requestAudioLauncher = registerForActivityResult(
        ActivityResultContracts.RequestPermission()
    ) { isGranted ->
        viewModel.onMicrophonePermissionResult(isGranted)
    }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)

        if (ContextCompat.checkSelfPermission(this, Manifest.permission.RECORD_AUDIO) == PackageManager.PERMISSION_GRANTED) {
            viewModel.onMicrophonePermissionResult(true)
        } else {
            requestAudioLauncher.launch(Manifest.permission.RECORD_AUDIO)
        }

        setContent {
            BujjiScreen(viewModel = viewModel, onRequestMicPermission = {
                requestAudioLauncher.launch(Manifest.permission.RECORD_AUDIO)
            })
        }
    }
}
